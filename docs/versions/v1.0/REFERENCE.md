Locked in: all-cloud accounts, consumers-only Microsoft, **full send mode** — so the allowlist stays in (it's the only control that doesn't depend on model obedience), `ro` token still exists for anything you don't trust with send, and `confirm:true` gates send at the API. Two spec deltas from my earlier pitch, both robustness calls:

1. **MCP ships as stdio (`mailhub mcp`), not mounted-in-daemon HTTP.** Mounting FastMCP inside FastAPI requires gluing two lifespans together across SDK versions — exactly the kind of seam that goes flaky on a library bump. stdio is bulletproof, works even if the hub is down, and both opencode and pi.dev handle it. The hub stays: it's what the CLI, REST surface, and audit log hang off.
2. **Hub binds 127.0.0.1 only** — agents run in the same WSL distro, so nothing touches your bridged LAN and there's no Windows Firewall rule to get wrong.

```python
import json, os, tempfile, threading, time
from .config import TOKENS_PATH

_lock = threading.Lock()

def _load():
    if not TOKENS_PATH.exists():
        return {}
    return json.loads(TOKENS_PATH.read_text())

def _save(tokens):
    TOKENS_PATH.parent.mkdir(parents=True, exist_ok=True)
    fd, tmp = tempfile.mkstemp(dir=TOKENS_PATH.parent, prefix=".tokens.")
    with os.fdopen(fd, "w") as f:
        json.dump(tokens, f, indent=1)
        f.flush()
        os.fsync(f.fileno())
    os.chmod(tmp, 0o600)
    os.replace(tmp, TOKENS_PATH)          # atomic: never a half-written token file

def get(alias):
    with _lock:
        return _load().get(alias)

def update(alias, **fields):
    with _lock:
        tokens = _load()
        t = tokens.setdefault(alias, {})
        t.setdefault("since", int(time.time()))   # first-consent time; survives refreshes
        t.update(fields)
        t["updated_at"] = int(time.time())
        _save(tokens)
        return t
```

```python
import base64, sys
from urllib.parse import urlencode, urlparse, parse_qs
import httpx

REDIRECT     = "http://localhost:8788"   # dead port → paste-back flow
GOOGLE_AUTH  = "https://accounts.google.com/o/oauth2/v2/auth"
GOOGLE_TOKEN = "https://oauth2.googleapis.com/token"
MS_AUTH      = "https://login.microsoftonline.com/consumers/oauth2/v2.0/authorize"
MS_TOKEN     = "https://login.microsoftonline.com/consumers/oauth2/v2.0/token"
GOOGLE_SCOPE = "https://www.googleapis.com/auth/gmail.modify"   # sensitive, not restricted; no hard-delete
MS_SCOPE     = "offline_access Mail.ReadWrite Mail.Send User.Read MailboxSettings.ReadWrite"

class ReauthNeeded(Exception):
    pass

def _state(provider, alias):
    return base64.urlsafe_b64encode(f"{provider}|{alias}".encode()).decode().rstrip("=")

def auth_url(provider, client_id, email, alias):
    common = dict(client_id=client_id, redirect_uri=REDIRECT, response_type="code",
                  login_hint=email, state=_state(provider, alias))
    if provider == "google":
        params = common | dict(scope=GOOGLE_SCOPE, access_type="offline", prompt="consent")
        return f"{GOOGLE_AUTH}?{urlencode(params)}"
    params = common | dict(scope=MS_SCOPE, prompt="consent")
    return f"{MS_AUTH}?{urlencode(params)}"

def parse_redirect(text):
    """Accepts the full failed localhost URL or a bare code."""
    text = text.strip()
    if text.startswith("http"):
        q = parse_qs(urlparse(text).query)
        return q.get("code", [None])[0], q.get("state", [None])[0]
    return text, None

def check_state(state, provider, alias):
    if state is not None and state != _state(provider, alias):
        sys.exit("state mismatch — the pasted URL is from a different auth attempt")

def _token_post(url, data):
    r = httpx.post(url, data=data, timeout=30)
    if r.status_code == 400 and "invalid_grant" in r.text:
        raise ReauthNeeded("token rejected by provider (revoked or expired)")
    r.raise_for_status()
    return r.json()

def exchange(provider, client_id, client_secret, code):
    if provider == "google":
        data = dict(code=code, client_id=client_id, client_secret=client_secret,
                    redirect_uri=REDIRECT, grant_type="authorization_code")
    else:
        data = dict(code=code, client_id=client_id, redirect_uri=REDIRECT,
                    grant_type="authorization_code", scope=MS_SCOPE)
    return _token_post(GOOGLE_TOKEN if provider == "google" else MS_TOKEN, data)

def refresh(provider, client_id, client_secret, refresh_token):
    if provider == "google":
        data = dict(client_id=client_id, client_secret=client_secret,
                    refresh_token=refresh_token, grant_type="refresh_token")
    else:
        data = dict(client_id=client_id, refresh_token=refresh_token,
                    grant_type="refresh_token", scope=MS_SCOPE)
    return _token_post(GOOGLE_TOKEN if provider == "google" else MS_TOKEN, data)
```

```python
import re
from email.message import EmailMessage
from html.parser import HTMLParser

class _TextExtractor(HTMLParser):
    SKIP  = {"script", "style", "head", "title"}
    BLOCK = {"p", "div", "br", "tr", "li", "h1", "h2", "h3", "h4", "table", "blockquote", "pre"}

    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.out, self.skip = [], 0

    def handle_starttag(self, tag, attrs):
        if tag in self.SKIP:   self.skip += 1
        elif tag in self.BLOCK: self.out.append("\n")

    def handle_endtag(self, tag):
        if tag in self.SKIP and self.skip: self.skip -= 1
        elif tag in self.BLOCK:            self.out.append("\n")

    def handle_data(self, data):
        if not self.skip:
            self.out.append(data)

def html_to_text(s):
    if not s:
        return ""
    p = _TextExtractor()
    p.feed(s)
    t = "".join(p.out)
    t = re.sub(r"[ \t]+", " ", t)
    return re.sub(r"\n{3,}", "\n\n", t).strip()

def split_addrs(x):
    items = x.split(",") if isinstance(x, str) else x
    return [a.strip() for a in (items or []) if a and a.strip()]

def build_mime(from_addr, to, subject, body, cc=None, html=False,
               in_reply_to=None, references=None):
    m = EmailMessage()
    m["From"], m["To"], m["Subject"] = from_addr, ", ".join(to), subject
    if cc:
        m["Cc"] = ", ".join(cc)
    m.set_content(body, subtype="html" if html else "plain")
    if in_reply_to:
        m["In-Reply-To"] = in_reply_to
        m["References"] = " ".join(x for x in (references, in_reply_to) if x)
    return m
```

```python
import asyncio, base64
from email.utils import parseaddr, parsedate_to_datetime
from .util import html_to_text, build_mime

API = "https://gmail.googleapis.com/gmail/v1/users/me"

def _b64d(s):
    return base64.urlsafe_b64decode(s + "=" * (-len(s) % 4))

class Gmail:
    def __init__(self, core, acct):
        self.core, self.acct = core, acct

    async def _api(self, method, path, **kw):
        return await self.core.api(self.acct.alias, method, f"{API}{path}", **kw)

    async def folders(self):
        r = await self._api("GET", "/labels")
        return [{"id": l["id"], "name": l["name"], "type": l["type"]} for l in r.get("labels", [])]

    async def _label_id(self, name):
        want = name.lower()
        for l in await self.folders():
            if want in (l["name"].lower(), l["id"].lower()):
                return l["id"]
        raise LookupError(f"gmail label not found: {name}")

    async def search(self, q=None, folder=None, limit=25):
        params = {"maxResults": limit}
        if q:
            params["q"] = q
        if folder:
            params["labelIds"] = await self._label_id(folder)
        r = await self._api("GET", "/messages", params=params)
        sem = asyncio.Semaphore(8)
        async def one(mid):
            async with sem:
                try:
                    return self._env(await self._api(
                        "GET", f"/messages/{mid}", params={"format": "metadata"}))
                except Exception:
                    return None
        msgs = await asyncio.gather(*(one(m["id"]) for m in r.get("messages", [])))
        return [m for m in msgs if m]

    def _env(self, m):
        hdr = {h["name"].lower(): h["value"] for h in m.get("payload", {}).get("headers", [])}
        name, addr = parseaddr(hdr.get("from", ""))
        try:
            date = parsedate_to_datetime(hdr["date"]).isoformat()
        except Exception:
            date = hdr.get("date", "")
        atts = []
        def collect(p):
            if p.get("filename") and p.get("body", {}).get("attachmentId"):
                atts.append(True)
            for c in p.get("parts", []):
                collect(c)
        collect(m.get("payload", {}))
        return {
            "account": self.acct.alias, "provider": "google",
            "id": m["id"], "thread_id": m.get("threadId"),
            "message_id": hdr.get("message-id", ""), "references": hdr.get("references", ""),
            "from": {"name": name, "addr": addr},
            "to": [parseaddr(a)[1] for a in (hdr.get("to") or "").split(",") if parseaddr(a)[1]],
            "subject": hdr.get("subject", ""), "date": date,
            "snippet": m.get("snippet", ""),
            "unread": "UNREAD" in m.get("labelIds", []),
            "has_attachments": bool(atts),
            "folder": m.get("labelIds", []),
        }

    def _bodies(self, payload):
        out = {}
        def walk(p):
            data = p.get("body", {}).get("data")
            mt = p.get("mimeType")
            if data and mt == "text/plain":
                out.setdefault("text", _b64d(data).decode("utf-8", "replace"))
            elif data and mt == "text/html":
                out.setdefault("html", _b64d(data).decode("utf-8", "replace"))
            for c in p.get("parts", []):
                walk(c)
        walk(payload)
        return out.get("text"), out.get("html")

    async def get(self, mid, html=False):
        m = await self._api("GET", f"/messages/{mid}", params={"format": "full"})
        env = self._env(m)
        text, body_html = self._bodies(m.get("payload", {}))
        if html:
            body = body_html or (text or "")
        else:
            body = text if text is not None else html_to_text(body_html or "")
        atts = []
        def collect(p):
            if p.get("filename") and p.get("body", {}).get("attachmentId"):
                atts.append({"id": p["body"]["attachmentId"], "filename": p["filename"],
                             "mime": p.get("mimeType", ""), "size": p["body"].get("size", 0)})
            for c in p.get("parts", []):
                collect(c)
        collect(m.get("payload", {}))
        env.update(body=body, attachments=atts)
        return env

    async def attachment(self, mid, aid):
        r = await self._api("GET", f"/messages/{mid}/attachments/{aid}")
        return _b64d(r["data"])

    async def send(self, to, subject, body, cc=None, html=False, reply_to_env=None):
        raw = build_mime(self.acct.email, to, subject, body, cc=cc, html=html,
                         in_reply_to=(reply_to_env or {}).get("message_id"),
                         references=(reply_to_env or {}).get("references"))
        payload = {"raw": base64.urlsafe_b64encode(raw.as_bytes()).decode()}
        if reply_to_env and reply_to_env.get("thread_id"):
            payload["threadId"] = reply_to_env["thread_id"]
        r = await self._api("POST", "/messages/send", json=payload)
        return {"sent": True, "id": r["id"], "thread_id": r.get("threadId")}

    async def draft(self, to, subject, body, cc=None, html=False, reply_to_env=None):
        raw = build_mime(self.acct.email, to, subject, body, cc=cc, html=html,
                         in_reply_to=(reply_to_env or {}).get("message_id"),
                         references=(reply_to_env or {}).get("references"))
        r = await self._api("POST", "/drafts", json={"message": {
            "raw": base64.urlsafe_b64encode(raw.as_bytes()).decode()}})
        return {"draft_id": r["id"], "message_id": r["message"]["id"]}

    async def mark(self, mid, read=True):
        payload = {"removeLabelIds": ["UNREAD"]} if read else {"addLabelIds": ["UNREAD"]}
        await self._api("POST", f"/messages/{mid}/modify", json=payload)
        return {"ok": True}

    async def move(self, mid, folder):
        lid = await self._label_id(folder)
        await self._api("POST", f"/messages/{mid}/modify", json={"addLabelIds": [lid]})
        return {"ok": True, "note": f"gmail: applied label '{lid}' (move = label in gmail)"}

    async def trash(self, mid):
        await self._api("POST", f"/messages/{mid}/trash")
        return {"ok": True}

    async def profile(self):
        return await self._api("GET", "/profile")
```

## mailhub/core.py

```python
import asyncio, fnmatch, json, time
from datetime import datetime, timezone
from pathlib import Path
import httpx
from . import oauth
from .config import accounts, AUDIT_PATH, STATE_DIR
from .store import get as token_get, update as token_update
from .gmail import Gmail
from .graph import Graph

class HubError(Exception):
    status = 500
class ReauthNeeded(HubError):
    status = 409
class ConfirmRequired(HubError):
    status = 428
class NotAllowed(HubError):
    status = 403
class NotFound(HubError):
    status = 404
class UpstreamError(HubError):
    status = 502

class Core:
    def __init__(self, cfg):
        self.cfg = cfg
        self.http = httpx.AsyncClient(timeout=45)
        self.accts = {a.alias: a for a in accounts(cfg)}
        self._locks = {}
        self._prov = {a.alias: (Gmail(self, a) if a.provider == "google" else Graph(self, a))
                      for a in self.accts.values()}

    def _lock(self, alias):
        return self._locks.setdefault(alias, asyncio.Lock())

    def _acct(self, alias):
        if alias not in self.accts:
            raise NotFound(f"unknown account alias '{alias}' — see /accounts")
        return self.accts[alias]

    # ---- tokens -----------------------------------------------------------

    def _tok(self, alias):
        t = token_get(alias)
        if not t or not t.get("refresh_token"):
            a = self._acct(alias)
            raise ReauthNeeded(f"no tokens for '{alias}' — run: mailhub auth {a.provider} {alias}")
        return t

    async def access(self, alias):
        t = self._tok(alias)
        if t.get("access_token") and t.get("expires_at", 0) > time.time() + 90:
            return t["access_token"]
        return (await self._renew(alias))["access_token"]

    async def _renew(self, alias):
        async with self._lock(alias):          # serializes refresh; MS rotates refresh tokens
            t = self._tok(alias)               # re-read inside the lock
            if t.get("access_token") and t.get("expires_at", 0) > time.time() + 90:
                return t
            a = self._acct(alias)
            try:
                r = oauth.refresh(a.provider, a.client["client_id"],
                                  a.client.get("client_secret"), t["refresh_token"])
            except oauth.ReauthNeeded:
                token_update(alias, status="needs_reauth")
                raise ReauthNeeded(f"'{alias}' refresh token rejected — run: mailhub auth {a.provider} {alias}")
            fields = {"access_token": r["access_token"],
                      "expires_at": int(time.time()) + int(r.get("expires_in", 3600)) - 120,
                      "status": "ok"}
            if r.get("refresh_token"):         # ms rotates; google omits → keep existing
                fields["refresh_token"] = r["refresh_token"]
            return token_update(alias, **fields)

    async def api(self, alias, method, url, *, params=None, json=None, headers=None, raw=False):
        for attempt in (1, 2):
            tok = await self.access(alias)
            h = {"Authorization": f"Bearer {tok}"}
            if headers:
                h.update(headers)
            r = await self.http.request(method, url, headers=h, params=params, json=json)
            if r.status_code == 401:
                await self._renew(alias)
                continue
            if r.status_code in (429, 503) and attempt == 1:
                try: wait = min(float(r.headers.get("Retry-After", "2")), 10)
                except ValueError: wait = 2
                await asyncio.sleep(wait)
                continue
            break
        if r.status_code >= 400:
            a = self._acct(alias)
            raise UpstreamError(f"{a.provider} {r.status_code}: {r.text[:300]}")
        if raw:
            return r.content
        return r.json() if r.content else {}

    # ---- domain -----------------------------------------------------------

    async def folders(self, alias):
        return await self._prov[self._acct(alias).alias].folders()

    async def search(self, alias, q=None, folder=None, limit=25):
        self._acct(alias)
        return await self._prov[alias].search(q=q, folder=folder, limit=min(limit, 50))

    async def get(self, alias, mid, html=False):
        self._acct(alias)
        return await self._prov[alias].get(mid, html=html)

    async def attachment(self, alias, mid, aid):
        self._acct(alias)
        return await self._prov[alias].attachment(mid, aid)

    async def mark(self, alias, mid, read=True):
        self._acct(alias)
        res = await self._prov[alias].mark(mid, read=read)
        self.audit("mark_read" if read else "mark_unread", alias, {"id": mid})
        return res

    async def move(self, alias, mid, folder):
        self._acct(alias)
        res = await self._prov[alias].move(mid, folder)
        self.audit("move", alias, {"id": mid, "folder": folder})
        return res

    async def trash(self, alias, mid):
        self._acct(alias)
        res = await self._prov[alias].trash(mid)
        self.audit("trash", alias, {"id": mid})
        return res

    async def draft(self, alias, to, subject, body, cc=None, html=False, reply_to=None):
        self._acct(alias)
        to, cc = self._addrs(to), self._addrs(cc)
        env = await self._reply_env(alias, reply_to)
        res = await self._prov[alias].draft(to, subject, body, cc=cc, html=html, reply_to_env=env)
        self.audit("draft", alias, {"to": to, "subject": subject})
        return res

    async def send(self, alias, to, subject, body, cc=None, html=False,
                   reply_to=None, confirm=False):
        self._acct(alias)
        to, cc = self._addrs(to), self._addrs(cc)
        self._allowlist_or_raise(alias, to + (cc or []))
        if not confirm:
            raise ConfirmRequired("dry run — resend with confirm=true to actually send")
        env = await self._reply_env(alias, reply_to)
        res = await self._prov[alias].send(to, subject, body, cc=cc, html=html, reply_to_env=env)
        self.audit("send", alias, {"to": to, "cc": cc, "subject": subject})
        return res

    def _addrs(self, x):
        from .util import split_addrs
        return split_addrs(x) if x else []

    def _allowlist_or_raise(self, alias, addrs):
        pats = (self.cfg.get("send") or {}).get("allowlist") or []
        if not pats:
            return
        for addr in addrs:
            if not any(fnmatch.fnmatch(addr.lower(), p.lower()) for p in pats):
                self.audit("send_blocked", alias, {"addr": addr})
                raise NotAllowed(f"'{addr}' is not in the send allowlist (config [send].allowlist)")

    async def _reply_env(self, alias, reply_to):
        if not reply_to:
            return None
        try:
            return await self.get(alias, reply_to)
        except HubError:
            return None

    # ---- status / audit ----------------------------------------------------

    def accounts_status(self):
        out = []
        for alias, a in self.accts.items():
            t = token_get(alias) or {}
            out.append({"alias": alias, "provider": a.provider, "email": a.email,
                        "status": t.get("status", "ok" if t else "no_tokens"),
                        "token_age_days": int((time.time() - t.get("since", time.time())) // 86400) if t else None})
        return out

    async def doctor(self):
        out = []
        for alias, a in self.accts.items():
            t = token_get(alias)
            if not t:
                out.append({"alias": alias, "provider": a.provider, "status": "no_tokens",
                            "hint": f"mailhub auth {a.provider} {alias}"})
                continue
            try:
                await self._prov[alias].profile()
                out.append({"alias": alias, "provider": a.provider, "status": "ok",
                            "token_age_days": int((time.time() - t.get("since", 0)) // 86400)})
            except ReauthNeeded as e:
                out.append({"alias": alias, "provider": a.provider, "status": "needs_reauth", "hint": str(e)})
            except UpstreamError as e:
                out.append({"alias": alias, "provider": a.provider, "status": "error", "detail": str(e)[:200]})
        return out

    def audit(self, action, alias, detail):
        AUDIT_PATH.parent.mkdir(parents=True, exist_ok=True)
        line = json.dumps({"ts": datetime.now(timezone.utc).isoformat(timespec="seconds"),
                           "action": action, "account": alias, **detail})
        fd = __import__("os").open(AUDIT_PATH, __import__("os").O_WRONLY | __import__("os").O_APPEND | __import__("os").O_CREAT, 0o600)
        __import__("os").write(fd, (line + "\n").encode())
        __import__("os").close(fd)
```

```python
def create_app(core: Core) -> FastAPI:
    auth_cfg = core.cfg.get("auth") or {}
    RO, FULL = auth_cfg.get("ro_token", ""), auth_cfg.get("full_token", "")

    app = FastAPI(title="mailhub", docs_url=None, redoc_url=None, openapi_url=None)

    @app.middleware("http")
    async def auth_middleware(request, call_next):
        if request.url.path == "/health":
            return await call_next(request)
        tok = request.headers.get("authorization", "").removeprefix("Bearer ").strip()
        if FULL and hmac.compare_digest(tok, FULL):
            request.state.scope = "full"
        elif RO and hmac.compare_digest(tok, RO):
            request.state.scope = "ro"
        else:
            return JSONResponse({"error": "bad or missing bearer token"}, status_code=403)
        return await call_next(request)

    def need_full(request: Request):
        if request.state.scope != "full":
            raise HTTPException(403, "full token required for mutations")

    app.add_exception_handler(HubError,        lambda r, e: error(e))
```

**4. Authorize each account** (paste-back flow, because the Windows browser's localhost isn't WSL's):

```bash
mailhub auth google me          # prints URL → open in Windows browser → consent ("unsafe" warning is expected)
                                # browser 404s at localhost:8788 → copy the address-bar URL →
mailhub auth google me --code 'http://localhost:8788/?code=4%2F0A...&state=...'
mailhub auth microsoft out --code '...'
mailhub doctor                  # every row must say ok
```

**5. Service it:**

```bash
mkdir -p ~/.config/systemd/user
cp ~/mailhub/deploy/*.{service,timer} ~/.config/systemd/user/
systemctl --user daemon-reload
systemctl --user enable --now mailhub.service mailhub-doctor.timer
loginctl enable-linger          # keeps the user manager alive
```

## First-trip checklist

- `mail accounts` → `token_age_days` climbing **past 7** proves you escaped the Google Testing trap.
- Try `mail search out "hasAttachments:true received>=2025-01-01"` and `mail search me "has:attachment newer_than:30d"` — confirm both dialects work so you can tell agents which account speaks which.
- Deliberately attempt an off-allowlist send once → expect 403 + a `send_blocked` line in `~/.local/state/mailhub/audit.log`.
- Keep `~/.config/mailhub/` and `~/.local/state/mailhub/` **out of any synced backup** — that's the token crown jewels.
- When an account shows `needs_reauth` (password change, manual revoke), it fails alone with a hint; the rest keep working. Fix with one `mailhub auth` rerun.
